<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Kategori extends Model
{
    public const TIPE = ['dimakan', 'pakan', 'kompos'];

    protected $guarded = [];

    protected $casts = [
        'aktif'        => 'boolean',
        'harga_per_kg' => 'integer',
    ];

    /** Kategori tujuan pengalihan (untuk kategori bertipe dimakan). */
    public function tujuan()
    {
        return $this->belongsTo(self::class, 'tujuan_id');
    }

    public function produks()
    {
        return $this->hasMany(Produk::class, 'kategori_id');
    }

    public function scopeAktif($query)
    {
        return $query->where('aktif', true);
    }

    public function scopeTujuanPengalihan($query)
    {
        return $query->where('aktif', true)
            ->whereIn('tipe', ['pakan', 'kompos']);
    }

    public function hasValidHarga(): bool
    {
        return $this->harga_per_kg !== null && $this->harga_per_kg > 0;
    }

    protected static function booted(): void
    {
        static::deleting(function (Kategori $kategori) {
            $dipakai = \Illuminate\Support\Facades\DB::table('produks')
                ->where('kategori_id', $kategori->id)
                ->orWhere('kategori_asli_id', $kategori->id)
                ->orWhere('target_kategori_id', $kategori->id)
                ->exists();

            if ($dipakai) {
                throw new \Exception("Kategori \"{$kategori->nama}\" tidak dapat dihapus karena masih digunakan oleh produk.");
            }

            if (static::where('tujuan_id', $kategori->id)->exists()) {
                throw new \Exception("Kategori \"{$kategori->nama}\" tidak dapat dihapus karena masih dijadikan tujuan pengalihan oleh kategori lain.");
            }
        });
    }
}

<?php

namespace App\Models;

class PengalihanKategori
{
    /**
     * Cadangan bila tabel kategoris belum ada / kosong.
     * Sumber utama sekarang tabel kategoris (diatur admin).
     */
    public const MAP = [
        'Roti & Pastry'        => ['tipe' => 'pakan',  'kategori' => 'Pakan Ternak & Ikan'],
        'Roti Tawar'           => ['tipe' => 'pakan',  'kategori' => 'Pakan Ternak & Ikan'],
        'Siap Santap'          => ['tipe' => 'pakan',  'kategori' => 'Pakan Ternak & Ikan'],
        'Sayur & Buah'         => ['tipe' => 'kompos', 'kategori' => 'Bahan Kompos & Pupuk'],
        'Bahan Olahan'         => ['tipe' => 'kompos', 'kategori' => 'Limbah Organik'],
        'Bahan & Olahan Dapur' => ['tipe' => 'kompos', 'kategori' => 'Limbah Organik'],
    ];

    /** Cari tujuan berdasarkan kategori asal (fallback ke default pakan). */
    public static function cariTujuan(string $kategoriAsal): array
    {
        try {
            $tujuan = Kategori::with('tujuan')->where('nama', $kategoriAsal)->first()?->tujuan;
            if ($tujuan) {
                return [
                    'tipe'         => $tujuan->tipe,
                    'kategori'     => $tujuan->nama,
                    'harga_per_kg' => (int) ($tujuan->harga_per_kg ?? 0),
                ];
            }
        } catch (\Throwable $e) {
            // tabel kategoris belum dimigrasi: pakai cadangan di bawah
        }

        $fallback = self::MAP[$kategoriAsal] ?? ['tipe' => 'pakan', 'kategori' => Produk::KATEGORI_EXPIRED];
        return array_merge($fallback, ['harga_per_kg' => 5000]);
    }

    /** Tipe (pakan/kompos) dari sebuah kategori tujuan, null bila tidak dikenal. */
    public static function tipeUntuk(string $kategori): ?string
    {
        try {
            $tipe = Kategori::where('nama', $kategori)->value('tipe');
            if ($tipe) {
                return $tipe;
            }
        } catch (\Throwable $e) {
        }

        foreach (self::semuaTujuan() as $t) {
            if ($t['kategori'] === $kategori) {
                return $t['tipe'];
            }
        }

        return null;
    }

    /** Daftar semua kategori tujuan pakan/kompos untuk dropdown Flutter. */
    public static function semuaTujuan(): array
    {
        try {
            $rows = Kategori::aktif()
                ->whereIn('tipe', ['pakan', 'kompos'])
                ->orderBy('urutan')->orderBy('nama')
                ->get(['id', 'nama', 'tipe', 'harga_per_kg']);

            if ($rows->isNotEmpty()) {
                return $rows->map(fn ($k) => [
                    'id'           => $k->id,
                    'tipe'         => $k->tipe,
                    'kategori'     => $k->nama,
                    'harga_per_kg' => (int) ($k->harga_per_kg ?? 0),
                ])->all();
            }
        } catch (\Throwable $e) {
        }

        return [
            ['id' => 1, 'tipe' => 'pakan',  'kategori' => 'Pakan Ternak & Ikan',  'harga_per_kg' => 5000],
            ['id' => 2, 'tipe' => 'kompos', 'kategori' => 'Bahan Kompos & Pupuk', 'harga_per_kg' => 3000],
            ['id' => 3, 'tipe' => 'kompos', 'kategori' => 'Ampas Kelapa & Kopi',  'harga_per_kg' => 2500],
            ['id' => 4, 'tipe' => 'kompos', 'kategori' => 'Limbah Organik',       'harga_per_kg' => 2000],
        ];
    }
}

    public const STATUS_PENGALIHAN_DIBUANG   = 'dibuang';

        'draf'                      => 'boolean',
        'rating'                    => 'float',
        'berat_kg'                  => 'float',
        'harga'                     => 'float',
        'harga_asli'                => 'float',
        'kadaluarsa'                => 'datetime',

    /**

     * Menyimpan harga_per_kg yang ditentukan ke kolom harga dan harga_asli.

    public function konfirmasiPengalihan(Kategori $kategori, float $beratAktual, ?int $hargaPerKg = null, ?string $catatan = null): self
    {
        $maxHarga = (int) ($kategori->harga_per_kg ?? 0);
        $hargaFinal = ($hargaPerKg !== null) ? min($maxHarga, max(0, $hargaPerKg)) : $maxHarga;

        $kategoriAsliId = $this->kategori_asli_id ?? $this->kategori_id;

        $surplus = self::create([
            'id_mitra'                   => $this->id_mitra,
            'nama'                       => 'Surplus ' . $kategori->nama . ' - ' . $this->nama,
            'deskripsi'                  => 'Bahan surplus dialihkan dari ' . $this->nama . ' (' . $beratAktual . ' kg). ' . ($catatan ?? 'Kondisi telah diperiksa mitra.'),

            'kategori'                   => $kategori->nama,
            'kategori_id'                => $kategori->id,
            'kategori_asli'              => $kategoriAsliNama,
            'kategori_asli_id'           => $kategoriAsliId,
            'gambar'                     => $this->gambar,
            'harga'                      => (int) $hargaFinal,
            'harga_asli'                 => (int) $hargaFinal,
            'satuan'                     => '/kg',
            'stok'                       => max(1, (int) floor($beratAktual)),
            'kuota'                      => max(1, (int) floor($beratAktual)),
            'berat_kg'                   => $beratAktual,
            'waktu_mulai'                => '00:00:00',
            'waktu_selesai'              => '23:59:00',
            'aktif'                      => true,
            'bersedia_dialihkan'         => false,
            'status_pengalihan'          => self::STATUS_PENGALIHAN_TIDAK,
            'created_at'                 => now(),
            'updated_at'                 => now(),
        ]);

        $this->update([
            'aktif'                      => false,
            'status_pengalihan'          => self::STATUS_PENGALIHAN_DIALIHKAN,
            'berat_aktual'               => $beratAktual,
            'target_kategori_pengalihan' => $kategori->nama,
            'target_kategori_id'         => $kategori->id,
            'produk_surplus_id'          => $surplus->id,
            'dialihkan_pada'             => now('Asia/Jakarta'),
            'dikonfirmasi_pada'          => now('Asia/Jakarta'),
            'catatan_pengalihan'         => $catatan,
        ]);

        return $surplus;
    }

    /**
     * Alihkan produk ini ke kategori tujuan secara manual.
     */
    public function alihkan(?string $kategoriTujuan = null, ?string $tipeTujuan = null): bool
    {
        if ($this->dialihkan_pada !== null) {
            return false;
        }

        $kategoriTujuanNama = $kategoriTujuan ?? $this->target_kategori_pengalihan;
        $katModel = null;
        if ($kategoriTujuanNama) {
            $katModel = Kategori::aktif()->where('nama', $kategoriTujuanNama)->first();
        }
        if (!$katModel) {
            $tujuan = PengalihanKategori::cariTujuan($this->kategori);
            $kategoriTujuanNama = $tujuan['kategori'];
            $katModel = Kategori::aktif()->where('nama', $kategoriTujuanNama)->first();
        }

        $tipe = $tipeTujuan ?? $katModel?->tipe ?? self::TIPE_PAKAN;
        $kategoriId = $katModel?->id;

        $this->update([
            'kategori_asli'      => $this->kategoriAsli?->nama ?? $this->kategori,
            'kategori_asli_id'   => $this->kategori_asli_id ?? $this->kategori_id,
            'kategori'           => $katModel?->nama ?? $kategoriTujuanNama,
            'kategori_id'        => $kategoriId,
            'target_kategori_id' => $kategoriId,
            'tipe'               => $tipe,
            'dialihkan_pada'     => now('Asia/Jakarta'),
            'status_pengalihan'  => self::STATUS_PENGALIHAN_DIALIHKAN,
        ]);

        return true;
    }

    public function getGambarUrlAttribute(): ?string
    {
        if (empty($this->gambar)) {
            return null;
        }

        $gambar = trim($this->gambar);
        if (str_starts_with($gambar, 'http://') || str_starts_with($gambar, 'https://')) {
            return $gambar;
        }

        $path = ltrim($gambar, '/');

        // Gunakan route API langsung /api/foto/{path} agar dapat diakses tanpa bergantung symlink webserver
        try {
            $root = request()->root();
            if (!empty($root)) {
                return rtrim($root, '/') . '/api/foto/' . $path;
            }
        } catch (\Throwable $e) {}

        return url('api/foto/' . $path);
    }

    /** Bentuk JSON yang cocok dengan field BahanItem di Flutter. */
    public function toApi(?float $lat = null, ?float $lng = null): array

        $gambarUrl = $this->gambar_url;

            'jarak_km'                  => $jarak,
            'kadaluarsa'                => $this->kadaluarsa

        ];
    }

    public static function haversine($lat1, $lng1, $lat2, $lng2): float
    {
        $r = 6371;
        $dLat = deg2rad($lat2 - $lat1);
        $dLng = deg2rad($lng2 - $lng1);
        $a = sin($dLat / 2) ** 2
            + cos(deg2rad($lat1)) * cos(deg2rad($lat2)) * sin($dLng / 2) ** 2;

        return 2 * $r * asin(sqrt($a));
    }

    public function getIsAnomaliAttribute(): bool
    {
        return $this->harga > $this->harga_asli;
    }

    public function scopeAnomali($query)
    {
        return $query->whereColumn('harga', '>', 'harga_asli');
    }

<?php

    public function index()
    {
        $kategoris = Kategori::with('tujuan')
            ->orderByRaw("FIELD(tipe, 'dimakan', 'pakan', 'kompos')")
            ->orderBy('urutan')
            ->orderBy('nama')
            ->get();

        // Jumlah produk per kategori
        $jumlahProdukById = DB::table('produks')
            ->select('kategori_id', DB::raw('COUNT(*) as total'))
            ->groupBy('kategori_id')
            ->pluck('total', 'kategori_id')
            ->all();

        $jumlahProdukByNama = DB::table('produks')
            ->select('kategori', DB::raw('COUNT(*) as total'))
            ->groupBy('kategori')
            ->pluck('total', 'kategori')
            ->all();

        $jumlahProduk = array_merge($jumlahProdukByNama, $jumlahProdukById);

        $opsiTujuan = Kategori::aktif()
            ->whereIn('tipe', ['pakan', 'kompos'])
            ->orderBy('urutan')->orderBy('nama')
            ->get();

        return view(self::VIEW, compact('kategoris', 'jumlahProduk', 'opsiTujuan'));
    }

    public function store(Request $request)
    {
        $data = $request->validate([
            'nama'         => ['required', 'string', 'max:100', 'unique:kategoris,nama'],
            'tipe'         => ['required', Rule::in(Kategori::TIPE)],
            'tujuan_id'    => ['nullable', 'integer', Rule::exists('kategoris', 'id')->whereIn('tipe', ['pakan', 'kompos'])],
            'harga_per_kg' => ['nullable', 'integer', 'min:0', 'max:99999999'],
        ], [
            'nama.required'    => 'Nama kategori wajib diisi.',
            'nama.unique'      => 'Nama kategori sudah dipakai.',
            'tipe.required'    => 'Pilih tipe kategori.',
            'tujuan_id.exists' => 'Tujuan pengalihan harus kategori pakan atau kompos.',
            'harga_per_kg.min' => 'Harga per kg minimal Rp 0.',
        ]);

        $hargaPerKg = in_array($data['tipe'], ['pakan', 'kompos']) ? (int) ($request->harga_per_kg ?? 0) : 0;

        Kategori::create([
            'nama'         => trim($data['nama']),
            'tipe'         => $data['tipe'],
            'tujuan_id'    => $data['tipe'] === 'dimakan' ? ($data['tujuan_id'] ?? null) : null,
            'harga_per_kg' => $hargaPerKg,
            'urutan'       => ((int) Kategori::max('urutan')) + 1,
            'aktif'        => true,
        ]);

        return back()->with('success', 'Kategori berhasil ditambahkan.');
    }

    public function update(Request $request, $id)
    {
        $kategori = Kategori::findOrFail($id);

        $data = $request->validate([
            'nama'         => ['required', 'string', 'max:100', Rule::unique('kategoris', 'nama')->ignore($kategori->id)],
            'urutan'       => ['nullable', 'integer', 'min:0', 'max:999'],
            'tujuan_id'    => ['nullable', 'integer', Rule::exists('kategoris', 'id')->whereIn('tipe', ['pakan', 'kompos'])],
            'harga_per_kg' => ['nullable', 'integer', 'min:0', 'max:99999999'],
        ], [
            'nama.unique'      => 'Nama kategori sudah dipakai.',
            'tujuan_id.exists' => 'Tujuan pengalihan harus kategori pakan atau kompos.',
            'harga_per_kg.min' => 'Harga per kg minimal Rp 0.',
        ]);

        $lama = $kategori->nama;
        $baru = trim($data['nama']);
        $hargaPerKg = in_array($kategori->tipe, ['pakan', 'kompos'])
            ? (int) ($request->harga_per_kg ?? $kategori->harga_per_kg ?? 0)
            : 0;

        DB::transaction(function () use ($kategori, $data, $lama, $baru, $hargaPerKg) {
            $kategori->update([
                'nama'         => $baru,
                'urutan'       => $data['urutan'] ?? $kategori->urutan,
                'tujuan_id'    => $kategori->tipe === 'dimakan' ? ($data['tujuan_id'] ?? null) : null,
                'harga_per_kg' => $hargaPerKg,
            ]);

            // Tetap perbarui kolom teks lama dengan satu query UPDATE per kolom selama masa transisi
            if ($lama !== $baru) {
                DB::table('produks')->where('kategori_id', $kategori->id)->update(['kategori' => $baru]);
                DB::table('produks')->where('kategori_asli_id', $kategori->id)->update(['kategori_asli' => $baru]);
                DB::table('produks')->where('target_kategori_id', $kategori->id)->update(['target_kategori_pengalihan' => $baru]);
            }
        });

        return back()->with('success', 'Kategori berhasil diperbarui.');
    }

    public function destroy($id)
    {
        $kategori = Kategori::findOrFail($id);

        $terpakaiProduk = DB::table('produks')
            ->where('kategori_id', $kategori->id)
            ->orWhere('kategori_asli_id', $kategori->id)
            ->orWhere('target_kategori_id', $kategori->id)
            ->exists();

        if ($terpakaiProduk) {
            return back()->with('error', "Kategori \"{$kategori->nama}\" tidak dapat dihapus karena masih digunakan oleh produk.");
        }

        if (Kategori::where('tujuan_id', $kategori->id)->exists()) {
            return back()->with('error', "Kategori \"{$kategori->nama}\" tidak dapat dihapus karena masih dijadikan tujuan pengalihan oleh kategori lain.");
        }

        try {
            $kategori->delete();
            return back()->with('success', "Kategori \"{$kategori->nama}\" berhasil dihapus.");
        } catch (\Illuminate\Database\QueryException $e) {
            return back()->with('error', "Kategori \"{$kategori->nama}\" tidak dapat dihapus karena masih memiliki keterkaitan data.");
        }
    }

    public function toggle($id)
    {
        $kategori = Kategori::findOrFail($id);
        $kategori->update(['aktif' => ! $kategori->aktif]);

        return back()->with('success', $kategori->aktif
            ? "Kategori {$kategori->nama} diaktifkan."
            : "Kategori {$kategori->nama} dinonaktifkan. Produk lama tetap aman.");
    }
}
